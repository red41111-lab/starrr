function doGet() { return HtmlService.createTemplateFromFile('index') .evaluate() .setTitle('별과 우주 탐구 대시보드') .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } function submitReport(data) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('응답') || ss.insertSheet('응답'); // 헤더가 없다면 생성 if (sheet.getLastRow() === 0) { sheet.appendRow(['응답 일시', '이름', '예상별의 등급기록', '오늘의 탐구결과', '피드백']); } sheet.appendRow([new Date(), data.name, data.score, data.report, '']); return true; } function getData() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('응답'); if (!sheet) return []; const values = sheet.getDataRange().getValues(); values.shift(); // 헤더 제거 // 배열 형태로 클라이언트에 전송 return values.map(row => ({ date: row[0], name: row[1], score: row[2], report: row[3], feedback: row[4] })); } function clearSheet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('응답'); if (sheet) { // 헤더 제외하고 내용 삭제 const lastRow = sheet.getLastRow(); if (lastRow > 1) { sheet.getRange(2, 1, lastRow - 1, 5).clearContent(); } } return true; } function saveFeedback(index, feedback) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('응답'); sheet.getRange(index + 2, 5).setValue(feedback); // 5번째 열에 피드백 저장 return true; }
